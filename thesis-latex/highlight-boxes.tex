\newsavebox{\highlightboxbody}
\newcommand{\mybox}[4]{%
  \begin{figure}[H]
    \centering
    \sbox{\highlightboxbody}{%
      \parbox{\dimexpr\linewidth-1.2cm\relax}{#4}}%
    \begin{tikzpicture}
      \node[
        anchor=north,
        draw=#2!70!black,
        rounded corners=3pt,
        line width=0.8pt,
        fill=#3,
        inner sep=5mm
      ] (big) {\raisebox{-4pt}[\ht\highlightboxbody][\dp\highlightboxbody]{%
        \usebox{\highlightboxbody}}};
      \node[
        draw=#2!70!black,
        rounded corners=3pt,
        line width=0.5pt,
        fill=#2,
        anchor=west,
        xshift=5mm,
        inner xsep=3mm,
        inner ysep=1.5mm
      ] at (big.north west) {\textbf{#1}};
    \end{tikzpicture}
  \end{figure}%
}
